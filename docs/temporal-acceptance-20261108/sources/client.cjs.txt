const {Client,Connection}=require('@temporalio/client');
const fs=require('node:fs');const assert=require('node:assert/strict');
const queue='sf-acceptance-20261108-2310',deploymentName=queue;
const registry={v1:{repository:'munichdeveloper/software-factory',factoryRelease:'9635a860260dda36b9bfc0215dd58b222fa33fd7',workflowVersion:'1',workflowHash:'test-workflow-v1'},v2:{repository:'munichdeveloper/sf-manager',factoryRelease:'53f35004559b9135f82f263e965cdac9cce18909',workflowVersion:'2',workflowHash:'test-workflow-v2'}};
const file='/tmp/evidence.json';let evidence=fs.existsSync(file)?JSON.parse(fs.readFileSync(file)):{};
const save=()=>{fs.writeFileSync(file,JSON.stringify(evidence,null,2));console.log(JSON.stringify(evidence));};
(async()=>{const connection=await Connection.connect({address:process.env.TEMPORAL_ADDRESS||'temporal:7233'});const client=new Client({connection,namespace:'software-factory-poc'});const handles={};for(const b of ['v1','v2'])handles[b]=client.workflow.getHandle(queue+'-'+b);
function validate(s,b){assert.deepEqual(s,registry[b],'UNKNOWN_OR_INCOMPATIBLE_SNAPSHOT');}
const phase=process.argv[2];
if(phase==='start'){for(const b of ['v1','v2']){const snapshot=registry[b];validate(snapshot,b);const h=await client.workflow.start('acceptance',{workflowId:queue+'-'+b,taskQueue:queue,args:[snapshot],versioningOverride:{pinnedTo:{deploymentName,buildId:b}}});evidence[b]={workflowId:h.workflowId,runId:h.firstExecutionRunId,snapshot};}for(const b of ['v1','v2']){for(let i=0;i<60;i++){try{const s=await handles[b].query('state');if(s.probe){evidence[b].paused=s;break;}}catch{} await new Promise(r=>setTimeout(r,500));}assert(evidence[b].paused);assert.equal(evidence[b].paused.probe.build,b);assert.equal(evidence[b].paused.probe.attempt,2);assert.deepEqual(evidence[b].paused.probe.credentialKeys,[]);}try{await client.workflow.start('acceptance',{workflowId:queue+'-v1',taskQueue:queue,args:[registry.v1]});throw new Error('DUPLICATE_ACCEPTED')}catch(e){assert.equal(e.name,'WorkflowExecutionAlreadyStartedError');evidence.duplicateDispatch='blocked';}for(const bad of [{...registry.v1,factoryRelease:'unknown'},{...registry.v1,workflowHash:'tampered'},{...registry.v1,repository:registry.v2.repository}]){assert.throws(()=>validate(bad,'v1'));}evidence.failClosed='unknown release, incompatible workflow hash, cross-repository snapshot rejected before dispatch';}
if(phase==='signal-stopped'){await handles.v1.signal('approve',{repository:registry.v2.repository,workflowHash:registry.v1.workflowHash,runKey:queue+'-v1',actor:'test-approver'});await handles.v1.signal('approve',{repository:registry.v1.repository,workflowHash:'stale',runKey:queue+'-v1',actor:'test-approver'});await handles.v1.signal('approve',{repository:registry.v1.repository,workflowHash:registry.v1.workflowHash,runKey:queue+'-v1',actor:'stranger'});await handles.v1.signal('approve',{repository:registry.v1.repository,workflowHash:registry.v1.workflowHash,runKey:queue+'-v1',actor:'test-approver'});await new Promise(r=>setTimeout(r,4000));const d=await handles.v1.describe();assert.equal(d.status.name,'RUNNING');assert.equal(d.runId,evidence.v1.runId);evidence.oldWorkerRequired=true;await handles.v2.signal('approve',{repository:registry.v2.repository,workflowHash:registry.v2.workflowHash,runKey:queue+'-v2',actor:'test-approver'});evidence.v2.result=await handles.v2.result();assert.equal(evidence.v2.result.probe.build,'v2');}
if(phase==='finish'){evidence.v1.result=await handles.v1.result();assert.equal(evidence.v1.result.probe.build,'v1');assert.equal(evidence.v1.result.rejected,3);assert.equal((await handles.v1.describe()).runId,evidence.v1.runId);evidence.recovery='same Run ID, original PINNED build v1, 3 invalid signals rejected';for(const b of ['v1','v2']){const history=await handles[b].fetchHistory();fs.writeFileSync('/tmp/history-'+b+'.json',JSON.stringify(history));const d=await handles[b].describe();evidence[b].versioningInfo=d.raw.workflowExecutionInfo.versioningInfo;evidence[b].eventCount=history.events.length;} }
save();await connection.close();})().catch(e=>{console.error(e);process.exitCode=1});
