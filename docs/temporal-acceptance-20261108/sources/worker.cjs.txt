const {Worker,NativeConnection}=require('@temporalio/worker');
const {Context}=require('@temporalio/activity');
const build=process.env.TEST_BUILD;
(async()=>{ const connection=await NativeConnection.connect({address:process.env.TEMPORAL_ADDRESS||'temporal:7233'}); const worker=await Worker.create({connection,namespace:'software-factory-poc',taskQueue:'sf-acceptance-20261108-2310',workflowsPath:require.resolve('./workflows-'+build+'.cjs'),workerDeploymentOptions:{version:{deploymentName:'sf-acceptance-20261108-2310',buildId:build},useWorkerVersioning:true,defaultVersioningBehavior:'PINNED'},activities:{async probe(snapshot){const attempt=Context.current().info.attempt;if(attempt===1)throw new Error('INJECTED_RETRY');return {build,attempt,repository:snapshot.repository,credentialKeys:Object.keys(process.env).filter(k=>/TOKEN|SECRET|PASSWORD|API_KEY/.test(k))};}},maxConcurrentActivityTaskExecutions:1,maxConcurrentWorkflowTaskExecutions:2});await worker.run();})().catch(e=>{console.error(e);process.exitCode=1});
