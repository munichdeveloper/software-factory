const {proxyActivities,defineSignal,defineQuery,setHandler,condition,workflowInfo}=require('@temporalio/workflow');const {publish}=proxyActivities({startToCloseTimeout:'20 seconds',heartbeatTimeout:'5 seconds',scheduleToCloseTimeout:'20 minutes',retry:{maximumAttempts:100,initialInterval:'1 second',maximumInterval:'5 seconds'}});exports.githubPublication=async function(s){let approved=false,rejected=0;setHandler(defineQuery('state'),()=>({approved,rejected,build:'b'}));setHandler(defineSignal('approve'),a=>{if(a.actor==='test-approver'&&a.repositoryId===s.repositoryId&&a.specHash===s.specHash&&a.workflowId===workflowInfo().workflowId)approved=true;else rejected++;});await condition(()=>approved);return {snapshot:s,rejected,...await publish(s)};};
